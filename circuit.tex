% ============================================================
%  NOT gate - transistor-level realization
%  A single common-emitter NPN stage.
%    Input 1 -> transistor ON  -> output pulled to GROUND (0)
%    Input 0 -> transistor OFF -> output pulled to +5 V   (1)
%  Compile:  pdflatex circuit.tex
% ============================================================
\documentclass[border=12pt]{standalone}
\usepackage{circuitikz}
\usepackage{amsmath}

% ground symbol (pic): same stroke as everything, just extra gap between bars
\tikzset{gnd/.pic={
  \draw
    (0,0) -- (0,-0.18)
    (-0.34,-0.18) -- (0.34,-0.18)
    (-0.21,-0.37) -- (0.21,-0.37)
    (-0.08,-0.56) -- (0.08,-0.56);
}}
\begin{document}
\begin{circuitikz}[american, line width=0.8pt]
  \ctikzset{tripoles/thickness=1, bipoles/thickness=1, logic ports/thickness=1}

  % ---------- Supply rail (+5 V) ----------
  \draw (-0.5,5) -- (6,5);
  \node[anchor=south] at (2.75,5.05) {\textbf{$V_{CC} = +5\,\mathrm{V}$}};
  \fill (2.5,5) circle (1.6pt);

  % ---------- transistor ----------
  \node[npn] (q) at (2.5,2) {};
  \node[anchor=west] at ($(q.E)+(0.45,0.15)$) {Q1 (2N3904)};
  % pin-name labels (match the physical 2N3904 legs)
  \node[red, font=\bfseries\small, anchor=south west] at ($(q.C)+(0.03,0.05)$) {C};
  \node[red, font=\bfseries\small, anchor=south east] at ($(q.B)+(-0.02,0.04)$) {B};
  \node[red, font=\bfseries\small, anchor=north west] at ($(q.E)+(0.05,-0.05)$) {E};

  % input + base resistor
  \draw (-1.7,2) node[ocirc]{} node[left]{Input $A$}
        to[R, l={$R_B=10\,\mathrm{k}\Omega$}] (q.B);

  % emitter -> ground
  \draw (q.E) -- ++(0,-1.1) pic{gnd};

  % collector pull-up to the rail
  \draw (q.C) to[R, l_={$R_C=1\,\mathrm{k}\Omega$}] (2.5,5);

  % output at the collector
  \draw (q.C) -- ++(2.7,0) node[ocirc]{} node[right]{Output $Y=\overline{A}$};

\end{circuitikz}
\end{document}
